cmake_minimum_required(VERSION 3.22.1)
project(polymath_llm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID AND NOT (ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_ABI STREQUAL "x86_64"))
    # Preserve installation of reading/notes/server chat on 32-bit phones; local inference is unavailable.
    add_library(polymath_llm SHARED unsupported.cpp)
    target_link_options(polymath_llm PRIVATE "-Wl,-z,max-page-size=16384")
    return()
endif()
set(LLAMA_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../../../.native/llama.cpp" CACHE PATH "Pinned llama.cpp checkout")
if(NOT EXISTS "${LLAMA_DIR}/include/llama.h")
    message(FATAL_ERROR "Run python scripts/prepare_local_llm.py from the repository root before building.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
# Portable baseline: do not require AVX or armv8.2 dot-product instructions.
set(GGML_AVX OFF CACHE BOOL "" FORCE)
set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
set(GGML_FMA OFF CACHE BOOL "" FORCE)
set(GGML_F16C OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
add_subdirectory("${LLAMA_DIR}" llama-build)
if(ANDROID)
    add_library(polymath_llm SHARED local_llm.cpp jni.cpp)
    target_link_libraries(polymath_llm PRIVATE llama log)
    target_link_options(polymath_llm PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
else()
    add_executable(polymath-local-smoke local_llm.cpp smoke.cpp)
    target_link_libraries(polymath-local-smoke PRIVATE llama)
endif()
